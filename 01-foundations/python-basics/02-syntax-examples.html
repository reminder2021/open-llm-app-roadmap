示例 1：创建和使用变量
# ============================================
# 示例 1：创建和使用变量
# ============================================

示例 5：类型转换
# ============================================
# 示例 5：类型转换
# ============================================

# 场景：LLM 返回的可能是字符串 "85"，需要转换成数字才能计算

# 字符串转整数
score_str = "85"                    # 这是一个字符串
score_int = int(score_str)          # 转换成整数
print("字符串:", score_str, type(score_str))  # 输出: 字符串: 85 
print("整数:", score_int, type(score_int))    # 输出: 整数: 85 

# 现在可以进行数学运算了
result = score_int + 15
print("85 + 15 =", result)          # 输出: 85 + 15 = 100

# 整数转字符串
age = 25
age_str = str(age)                  # 转换成字符串
print("我是" + age_str + "岁")      # 输出: 我是25岁

# 字符串转浮点数
price_str = "9.99"
price_float = float(price_str)      # 转换成浮点数
print("价格:", price_float)         # 输出: 价格: 9.99
print("两件价格:", price_float * 2) # 输出: 两件价格: 19.98

# 注意：不是所有字符串都能转换
# int("hello")  # 这会报错！"hello" 不能转换成整数
⚠️ 注意：只有数字格式的字符串才能转换成数字。比如 int("85") 可以，但 int("hello") 会报错。

示例 6：字符串拼接和格式化
# ============================================
# 示例 6：字符串拼接和格式化
# ============================================

# 方法一：用 + 号拼接（不推荐，麻烦）
first_name = "张"
last_name = "三"
full_name = first_name + last_name
print("拼接:", full_name)  # 输出: 拼接: 张三

# 方法二：用 f-string 格式化（推荐！）
name = "小明"
age = 25
message = f"我叫{name}，今年{age}岁"
print("f-string:", message)  # 输出: f-string: 我叫小明，今年25岁

# f-string 可以放表达式
price = 9.99
quantity = 3
total = f"总价: {price * quantity} 元"
print(total)  # 输出: 总价: 29.97 元

# f-string 可以格式化数字
score = 85.6789
print(f"分数: {score:.2f}")  # 输出: 分数: 85.68（保留2位小数）
print(f"分数: {score:.0f}")  # 输出: 分数: 86（取整）

# 多行 f-string
report = f"""
=== 测试报告 ===
项目名称: LLM 应用
测试人员: {name}
测试分数: {score:.1f}
是否通过: {'是' if score >= 60 else '否'}
================
"""
print(report)
💡 小贴士：f-string 是 Python 3.6 引入的格式化方式，用 f"..." 表示，花括号 {} 里可以放变量和表达式。
示例 7：常用字符串方法
# ============================================
# 示例 7：常用字符串方法
# ============================================

text = "  Hello, World!  "

# 1. 去掉首尾空格
clean_text = text.strip()

print(f"大写: {text.upper()}")      # 输出: 大写:   HELLO, WORLD!

has_world = "World" in text
print(f"是否包含 World: {has_world}")  # 输出: 是否包含 World: True

示例 8：构造 Prompt 模板（LLM 应用场景）
# ============================================
# 示例 8：构造 Prompt 模板（LLM 应用场景）
# ============================================

# 在 LLM 应用中，我们经常需要用模板构造 Prompt

# 用户的问题
user_question = "什么是 RAG？"

# 知识库检索到的内容
context = "RAG（检索增强生成）是一种结合检索和生成的技术。它先从知识库中检索相关信息，然后让模型基于这些信息生成回答。"

# 构造 Prompt 模板
prompt_template = """请根据以下参考资料回答用户的问题。

参考资料：
{context}

用户问题：
{question}

要求：
1. 只基于参考资料回答
2. 如果资料中没有答案，请回答"根据当前资料无法判断"
3. 回答要简洁明了
"""

# 用 f-string 填充模板
prompt = prompt_template.format(context=context, question=user_question)

# 打印最终的 Prompt
print("=== 生成的 Prompt ===")
print(prompt)
运行结果：
=== 生成的 Prompt ===
请根据以下参考资料回答用户的问题。

参考资料：
RAG（检索增强生成）是一种结合检索和生成的技术。它先从知识库中检索相关信息，然后让模型基于这些信息生成回答。

用户问题：
什么是 RAG？

要求：
1. 只基于参考资料回答
2. 如果资料中没有答案，请回答"根据当前资料无法判断"
3. 回答要简洁明了

示例 9：if-elif-else 条件判断
# ============================================
# 示例 9：if-elif-else 条件判断
# ============================================

# 场景：根据分数判断等级
score = 85

# if-elif-else 结构
# Python 用缩进（通常是4个空格）来表示代码块
if score >= 90:
    # 条件1：分数 >= 90
    grade = "优秀"
    print(f"分数 {score} → 等级: {grade}")
elif score >= 80:
    # 条件2：分数 >= 80（且 < 90）
    grade = "良好"
    print(f"分数 {score} → 等级: {grade}")
elif score >= 60:
    # 条件3：分数 >= 60（且 < 80）
    grade = "及格"
    print(f"分数 {score} → 等级: {grade}")
else:
    # 其他情况：分数 < 60
    grade = "不及格"
    print(f"分数 {score} → 等级: {grade}")
运行结果：
分数 85 → 等级: 良好
示例 10：逻辑运算符（and, or, not）

# 示例 10：逻辑运算符（and, or, not）
# ============================================

age = 25

score = 85

# and：两个条件都满足才为 True
if age >= 18 and has_id:
    print("✅ 年龄和身份证都满足，可以进入")  # 会执行

# or：至少一个条件满足就为 True

else:

# not：取反
is_vip = False
if not is_vip:
    print("普通用户")  # 会执行（因为 is_vip 是 False）

# 组合使用
if (age >= 18 and has_id) or is_vip:
    print("✅ 允许访问")  # 会执行

示例 13：while 循环和 break
# ============================================
# 示例 13：while 循环和 break
# ============================================

# while 循环：当条件为 True 时重复执行
count = 0
print("=== while 循环 ===")
while count < 5:
    print(f"第 {count + 1} 次循环")
    count += 1  # 等同于 count = count + 1

# break：提前退出循环
print("\n=== break 示例 ===")
for i in range(10):
    if i == 5:
        print("遇到 5，退出循环")
        break  # 立即退出循环
    print(f"i = {i}")

# continue：跳过本次循环
print("\n=== continue 示例 ===")
for i in range(5):
    if i == 2:
        print(f"跳过 i = {i}")
        continue  # 跳过本次循环，继续下一次
    print(f"i = {i}")
运行结果：

示例 14：批量处理数据（LLM 应用场景）
# ============================================
# 示例 14：批量处理数据（LLM 应用场景）
# ============================================

# 场景：批量处理用户问题，统计通过率

# 测试结果列表
test_results = [
    {"case_id": "TC001", "result": "通过", "score": 85},
    {"case_id": "TC002", "result": "失败", "score": 45},
    {"case_id": "TC003", "result": "通过", "score": 92},
    {"case_id": "TC004", "result": "通过", "score": 78},
    {"case_id": "TC005", "result": "失败", "score": 55},
]

# 统计通过率
total = len(test_results)      # 总数
passed = 0                      # 通过数
failed = 0                      # 失败数
total_score = 0                 # 总分

print("=== 测试结果统计 ===")
for result in test_results:
    # 打印每个结果
    print(f"{result['case_id']}: {result['result']} (分数: {result['score']})")
    
    # 统计
    if result["result"] == "通过":
        passed += 1
    else:
        failed += 1
    
    total_score += result["score"]

# 计算平均分
average_score = total_score / total

# 打印统计结果
print(f"\n=== 统计汇总 ===")
print(f"总数: {total}")
print(f"通过: {passed}")
print(f"失败: {failed}")
print(f"通过率: {passed/total:.1%}")
print(f"平均分: {average_score:.1f}")
运行结果：

TC002: 失败 (分数: 45)
TC003: 通过 (分数: 92)

示例 15：列表推导式
# ============================================
# 示例 15：列表推导式
# ============================================

# 列表推导式是一种简洁的创建列表的方式

# 普通方式：创建 1-10 的平方数列表
squares = []
for i in range(1, 11):
    squares.append(i * i)
print("普通方式:", squares)

# 列表推导式：更简洁
squares = [i * i for i in range(1, 11)]
print("列表推导式:", squares)

# 带条件的列表推导式：只要偶数
even_squares = [i * i for i in range(1, 11) if i % 2 == 0]
print("偶数的平方:", even_squares)

# 字符串处理示例
