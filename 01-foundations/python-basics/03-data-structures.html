列表 (List)

列表是有序的数据集合，可以存储多个值。用方括号 [] 创建。

创建列表

# 创建列表
fruits = ["苹果", "香蕉", "橙子"]
numbers = [1, 2, 3, 4, 5]
mixed = ["hello", 42, 3.14, True]  # 可以混合不同类型

# 空列表
empty = []

print(fruits)  # 输出: ['苹果', '香蕉', '橙子']

访问元素

fruits = ["苹果", "香蕉", "橙子", "葡萄", "西瓜"]

# 用索引访问（从 0 开始）
print(fruits[0])   # 输出: 苹果
print(fruits[2])   # 输出: 橙子

# 负数索引（从末尾开始）
print(fruits[-1])  # 输出: 西瓜（最后一个）
print(fruits[-2])  # 输出: 葡萄（倒数第二个）

# 切片
print(fruits[1:3])  # 输出: ['香蕉', '橙子']（索引 1 到 2）
print(fruits[:3])   # 输出: ['苹果', '香蕉', '橙子']（前 3 个）
print(fruits[2:])   # 输出: ['橙子', '葡萄', '西瓜']（从索引 2 开始）

修改列表

fruits = ["苹果", "香蕉", "橙子"]

# 修改元素
fruits[0] = "草莓"
print(fruits)  # 输出: ['草莓', '香蕉', '橙子']

# 添加元素
fruits.append("葡萄")  # 在末尾添加
print(fruits)  # 输出: ['草莓', '香蕉', '橙子', '葡萄']

# 插入元素
fruits.insert(1, "西瓜")  # 在索引 1 的位置插入
print(fruits)  # 输出: ['草莓', '西瓜', '香蕉', '橙子', '葡萄']

# 删除元素
fruits.remove("香蕉")  # 删除指定值
print(fruits)  # 输出: ['草莓', '西瓜', '橙子', '葡萄']

# 弹出元素（获取并删除）
last = fruits.pop()  # 弹出最后一个
print(last)    # 输出: 葡萄
print(fruits)  # 输出: ['草莓', '西瓜', '橙子']

列表操作

numbers = [3, 1, 4, 1, 5, 9, 2, 6]

# 长度
print(len(numbers))  # 输出: 8

# 排序
numbers.sort()
print(numbers)  # 输出: [1, 1, 2, 3, 4, 5, 6, 9]

# 反转
numbers.reverse()
print(numbers)  # 输出: [9, 6, 5, 4, 3, 2, 1, 1]

# 计数
print(numbers.count(1))  # 输出: 2

# 查找位置
print(numbers.index(5))  # 输出: 2

# 判断是否存在
print(5 in numbers)   # 输出: True
print(10 in numbers)  # 输出: False

遍历列表

fruits = ["苹果", "香蕉", "橙子"]

# 方式一：直接遍历
for fruit in fruits:
    print(fruit)

# 方式二：带索引遍历
for i, fruit in enumerate(fruits):
    print(f"{i}: {fruit}")

# 方式三：用 range
for i in range(len(fruits)):
    print(f"{i}: {fruits[i]}")

列表推导式

# 创建一个数字列表
numbers = [1, 2, 3, 4, 5]

# 每个数字乘以 2
doubled = [x * 2 for x in numbers]
print(doubled)  # 输出: [2, 4, 6, 8, 10]

# 只保留偶数
evens = [x for x in numbers if x % 2 == 0]
print(evens)  # 输出: [2, 4]

# 字符串处理
words = ["hello", "world", "python"]
upper_words = [w.upper() for w in words]
print(upper_words)  # 输出: ['HELLO', 'WORLD', 'PYTHON']

字典 (Dictionary)

字典是键值对的数据结构，用花括号 {} 创建。每个元素都有一个"键"和对应的"值"。

创建字典

# 创建字典
person = {
    "name": "小明",
    "age": 25,
    "city": "北京"
}

# 空字典
empty = {}

# 访问值
print(person["name"])  # 输出: 小明
print(person["age"])   # 输出: 25

# 用 get 方法（更安全，键不存在时返回默认值）
print(person.get("name"))        # 输出: 小明
print(person.get("phone", "无")) # 输出: 无（键不存在）

修改字典

person = {"name": "小明", "age": 25}

# 修改值
person["age"] = 26
print(person)  # 输出: {'name': '小明', 'age': 26}

# 添加新键值对
person["city"] = "北京"
print(person)  # 输出: {'name': '小明', 'age': 26, 'city': '北京'}

# 删除键值对
del person["city"]
print(person)  # 输出: {'name': '小明', 'age': 26}

# 使用 pop 删除并返回值
age = person.pop("age")
print(age)     # 输出: 26
print(person)  # 输出: {'name': '小明'}

字典操作

person = {"name": "小明", "age": 25, "city": "北京"}

# 获取所有键
print(person.keys())   # 输出: dict_keys(['name', 'age', 'city'])

# 获取所有值
print(person.values()) # 输出: dict_values(['小明', 25, '北京'])

# 获取所有键值对
print(person.items())  # 输出: dict_items([('name', '小明'), ('age', 25), ('city', '北京')])

# 判断键是否存在
print("name" in person)    # 输出: True
print("phone" in person)   # 输出: False

# 获取长度
print(len(person))  # 输出: 3

遍历字典

person = {"name": "小明", "age": 25, "city": "北京"}

# 遍历键
for key in person:
    print(key)

# 遍历值
for value in person.values():
    print(value)

# 遍历键值对
for key, value in person.items():
    print(f"{key}: {value}")
与 LLM 应用的关联：字典是 LLM 应用中最常用的数据结构，因为 API 的请求和响应通常都是字典格式。

    "messages": [
        {"role": "user", "content": "什么是 RAG？"}
    ],
    "temperature": 0.7
}

# LLM API 响应
response = {
    "id": "chatcmpl-123",
    "choices": [
        {
            "message": {
                "content": "RAG 是检索增强生成..."
            }
        }
    ],
    "usage": {
        "prompt_tokens": 10,
        "completion_tokens": 50
    }
}

嵌套结构

字典和列表可以嵌套使用，形成复杂的数据结构。这在处理 LLM 返回数据时非常常见。

列表嵌套字典

# 用户列表
users = [
    {"name": "张三", "age": 25},
    {"name": "李四", "age": 30},
    {"name": "王五", "age": 28}
]

# 访问第一个用户的名字
print(users[0]["name"])  # 输出: 张三

# 遍历所有用户
for user in users:
    print(f"{user['name']}，{user['age']}岁")

字典嵌套列表

# 学生成绩
student = {
    "name": "张三",
    "scores": [85, 92, 78, 91]
}

# 访问第一个成绩
print(student["scores"][0])  # 输出: 85

# 计算平均分
average = sum(student["scores"]) / len(student["scores"])
print(f"平均分: {average}")  # 输出: 平均分: 86.5

字典嵌套字典

# API 响应
response = {
    "status": 200,
    "data": {
        "user": {
            "name": "张三",
            "email": "zhangsan@example.com"
        },
        "token": "abc123"
    }
}

# 访问嵌套的值
print(response["data"]["user"]["name"])  # 输出: 张三

复杂嵌套示例

# 模拟 LLM API 响应
llm_response = {
    "id": "chatcmpl-123",
    "object": "chat.completion",
    "choices": [
        {
            "index": 0,
            "message": {
                "role": "assistant",
                "content": "RAG（检索增强生成）是一种结合检索和生成的技术..."
            },
            "finish_reason": "stop"
        }
    ],
    "usage": {
        "prompt_tokens": 15,
        "completion_tokens": 50,
        "total_tokens": 65
    }
}

# 提取回答内容
answer = llm_response["choices"][0]["message"]["content"]
print(answer)

# 提取 token 使用量
total_tokens = llm_response["usage"]["total_tokens"]
print(f"总共使用了 {total_tokens} 个 token")
访问嵌套数据的技巧：如果不确定某个键是否存在，可以使用 .get() 方法避免报错。 
# 安全地访问嵌套数据
answer = llm_response.get("choices", [{}])[0].get("message", {}).get("content", "")
print(answer)

JSON

JSON 是一种轻量级的数据交换格式，在 LLM 应用中无处不在。Python 可以轻松地处理 JSON 数据。

什么是 JSON？

JSON（JavaScript Object Notation）是一种文本格式，看起来很像 Python 的字典：

{
  "name": "小明",
  "age": 25,
  "skills": ["Python", "LLM", "RAG"]
}

JSON 和 Python 的对应关系

JSON → Python

  • 对象 {} → 字典 {}
  • 数组 [] → 列表 []
  • 字符串 "" → 字符串 ""
  • 数字 → 整数或浮点数
  • true/false → True/False
  • null → None

Python → JSON

  • 字典 {} → 对象 {}
  • 列表 [] → 数组 []
  • 字符串 "" → 字符串 ""
  • 整数/浮点数 → 数字
  • True/False → true/false
  • None → null

解析 JSON（JSON → Python）

import json

# JSON 字符串
json_string = '{"name": "小明", "age": 25, "skills": ["Python", "LLM"]}'

# 解析 JSON
data = json.loads(json_string)

# 现在 data 是一个 Python 字典
print(data)           # 输出: {'name': '小明', 'age': 25, 'skills': ['Python', 'LLM']}
print(data["name"])   # 输出: 小明
print(data["skills"]) # 输出: ['Python', 'LLM']

生成 JSON（Python → JSON）

import json

# Python 字典
data = {
    "name": "小明",
    "age": 25,
    "skills": ["Python", "LLM"]
}

# 转换成 JSON 字符串
json_string = json.dumps(data, ensure_ascii=False, indent=2)
print(json_string)

# 输出:
# {
#   "name": "小明",
#   "age": 25,
#   "skills": [
#     "Python",
#     "LLM"
#   ]
# }

参数说明

import json

data = {"name": "小明", "age": 25}

# ensure_ascii=False：让中文正常显示，不转成 \uXXXX
print(json.dumps(data, ensure_ascii=False))
# 输出: {"name": "小明", "age": 25}

# indent=2：格式化输出，缩进 2 个空格
print(json.dumps(data, ensure_ascii=False, indent=2))
# 输出:
# {
#   "name": "小明",
#   "age": 25
# }

处理 JSON 错误

import json

# 无效的 JSON 字符串
bad_json = '{"name": "小明", "age": 25,}'  # 注意末尾的逗号

try:
    data = json.loads(bad_json)
except json.JSONDecodeError as e:
    print(f"JSON 解析错误: {e}")
与 LLM 应用的关联：LLM 的 API 请求和响应都是 JSON 格式，你需要熟练地解析和生成 JSON。 
import json

# 构造 LLM API 请求
request = {
    "model": "qwen2.5:7b",
    "messages": [
        {"role": "user", "content": "什么是 RAG？"}
    ]
}

# 转换成 JSON 发送给 API
request_json = json.dumps(request)
print(request_json)

# 解析 API 响应
response_json = '{"choices": [{"message": {"content": "RAG 是..."}}]}'
response = json.loads(response_json)
answer = response["choices"][0]["message"]["content"]
print(answer)
