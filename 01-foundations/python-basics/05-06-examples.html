文本文件示例

示例 1：读写文本文件
# ============================================
# 示例 1：读写文本文件
# ============================================
from pathlib import Path

# 写入文本文件
content = """第一行：Hello
第二行：World
第三行：Python"""

# write_text() 会自动创建文件
Path("output.txt").write_text(content, encoding="utf-8")
print("文件已写入")

# 读取文本文件
read_content = Path("output.txt").read_text(encoding="utf-8")
print("文件内容:")
print(read_content)

# 逐行读取
lines = Path("output.txt").read_text(encoding="utf-8").splitlines()
print(f"共 {len(lines)} 行")
for i, line in enumerate(lines):
    print(f"  第 {i+1} 行: {line}")
运行结果：
文件已写入
文件内容:
第一行：Hello
第二行：World
第三行：Python

共 3 行
  第 1 行: 第一行：Hello
  第 2 行: 第二行：World
  第 3 行: 第三行：Python
示例 2：使用 with 语句（推荐）
# ============================================
# 示例 2：使用 with 语句（推荐）
# ============================================

# with 语句会自动关闭文件，更安全
# 写入文件
with open("output.txt", "w", encoding="utf-8") as f:
    f.write("第一行\n")
    f.write("第二行\n")
    f.write("第三行\n")
print("写入完成")

# 读取文件
with open("output.txt", "r", encoding="utf-8") as f:
    content = f.read()
    print("读取内容:")
    print(content)

# 逐行读取
with open("output.txt", "r", encoding="utf-8") as f:
    for line_num, line in enumerate(f, 1):
        print(f"  第 {line_num} 行: {line.strip()}")  # strip() 去掉换行符
💡 小贴士：with 语句会自动关闭文件，即使发生错误也会关闭。推荐总是使用 with 语句来操作文件。

JSON文件示例

示例 4：读写JSON文件
# ============================================
# 示例 4：读写JSON文件
# ============================================
import json
from pathlib import Path

# 准备数据
config = {
    "model": "qwen2.5:7b",
    "temperature": 0.7,
    "max_tokens": 512,
    "features": ["RAG", "Agent", "微调"]
}

# 写入 JSON 文件
Path("config.json").write_text(
    json.dumps(config, ensure_ascii=False, indent=2),
    encoding="utf-8"
)
print("配置已保存到 config.json")

# 读取 JSON 文件
content = Path("config.json").read_text(encoding="utf-8")
loaded_config = json.loads(content)

print("读取的配置:")
print(f"  模型: {loaded_config['model']}")
print(f"  温度: {loaded_config['temperature']}")
print(f"  功能: {loaded_config['features']}")
运行结果：

  温度: 0.7

示例 5：JSON Lines 格式（微调数据）
# ============================================
# 示例 5：JSON Lines 格式（微调数据）
# ============================================
import json

# JSONL 格式：每行一个 JSON 对象
# 常用于存储微调数据和评估数据集

# 写入 JSONL
samples = [
    {"instruction": "什么是RAG？", "output": "RAG是检索增强生成"},
    {"instruction": "什么是LLM？", "output": "LLM是大语言模型"},
    {"instruction": "什么是Agent？", "output": "Agent是智能代理"},
]

with open("train_data.jsonl", "w", encoding="utf-8") as f:
    for sample in samples:
        # ensure_ascii=False 让中文正常显示
        f.write(json.dumps(sample, ensure_ascii=False) + "\n")
print("训练数据已保存")

# 读取 JSONL
print("\n读取训练数据:")
with open("train_data.jsonl", "r", encoding="utf-8") as f:
    for line_num, line in enumerate(f, 1):
        data = json.loads(line.strip())
        print(f"  第{line_num}条: {data['instruction']} -> {data['output']}")
💡 小贴士：JSONL 格式常用于存储 LLM 微调数据，每行一个训练样本。

CSV文件示例

示例 6：读写CSV文件
# ============================================
# 示例 6：读写CSV文件
# ============================================
import csv

# 准备数据
test_results = [
    ["case_id", "question", "score", "result"],
    ["TC001", "什么是RAG？", 95, "通过"],
    ["TC002", "什么是LLM？", 88, "通过"],
    ["TC003", "什么是Agent？", 55, "失败"],
]

# 写入 CSV
with open("test_results.csv", "w", encoding="utf-8", newline="") as f:
    writer = csv.writer(f)
    writer.writerows(test_results)
print("测试结果已保存到 test_results.csv")

# 读取 CSV（使用 DictReader 更方便）
print("\n读取测试结果:")
with open("test_results.csv", "r", encoding="utf-8") as f:
    reader = csv.DictReader(f)
    for row in reader:
        status = "✅" if row["result"] == "通过" else "❌"
        print(f"  {status} {row['case_id']}: {row['question']} - {row['score']}分")
运行结果：
测试结果已保存到 test_results.csv

读取测试结果:
  ✅ TC001: 什么是RAG？ - 95分
  ✅ TC002: 什么是LLM？ - 88分
  ❌ TC003: 什么是Agent？ - 55分

示例 8：安全的文件读取
# ============================================
# 示例 8：安全的文件读取
# ============================================
import json

def safe_read_json(file_path):
    """安全地读取 JSON 文件
    
    Args:
        file_path: 文件路径
    
    Returns:
        解析后的数据，如果失败返回空字典
    """
    try:
        with open(file_path, "r", encoding="utf-8") as f:
            return json.load(f)
    except FileNotFoundError:
        print(f"文件不存在: {file_path}")
        return {}
    except json.JSONDecodeError as e:
        print(f"JSON 解析错误: {e}")
        return {}
    except Exception as e:
        print(f"未知错误: {e}")
        return {}

# 测试
config = safe_read_json("config.json")
print(f"配置: {config}")

config2 = safe_read_json("nonexistent.json")
print(f"不存在的文件: {config2}")
示例 9：带重试的函数调用
# ============================================
# 示例 9：带重试的函数调用
# ============================================
import random

def simulate_api_call():
    """模拟 API 调用（可能失败）"""
    if random.random() < 0.7:  # 70% 概率失败
        raise ConnectionError("网络连接失败")
    return {"status": "success", "data": "API 响应数据"}

def call_with_retry(max_retries=3):
    """带重试的 API 调用
    
    Args:
        max_retries: 最大重试次数
    
    Returns:
        API 响应，如果全部失败返回 None
    """
    for attempt in range(max_retries):
        try:
            result = simulate_api_call()
            print(f"  第 {attempt + 1} 次尝试成功！")
            return result
        except ConnectionError as e:
            print(f"  第 {attempt + 1} 次尝试失败: {e}")
            if attempt == max_retries - 1:
                print("  重试次数用完，放弃")
                return None

# 测试
print("调用 API:")
result = call_with_retry()
print(f"结果: {result}")
运行结果（示例）：
调用 API:
  第 1 次尝试失败: 网络连接失败
  第 2 次尝试失败: 网络连接失败
  第 3 次尝试成功！
结果: {'status': 'success', 'data': 'API 响应数据'}
💡 小贴士：在 LLM 应用中，网络请求可能失败，使用重试机制可以提高程序的健壮性。
示例 10：finally 清理资源
# ============================================
# 示例 10：finally 清理资源
# ============================================

def read_file_with_cleanup(file_path):
    """读取文件，确保资源被清理"""
    f = None
    try:
        f = open(file_path, "r", encoding="utf-8")
        content = f.read()
        return content
    except FileNotFoundError:
        print(f"文件不存在: {file_path}")
        return None
    finally:
        # finally 无论如何都会执行
        if f:
            f.close()
            print("文件已关闭")

# 测试
content = read_file_with_cleanup("test.txt")
print(f"内容: {content}")
