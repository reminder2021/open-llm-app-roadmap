Path 对象

Python 的 pathlib 模块提供了面向对象的文件路径操作，比传统的字符串拼接更方便、更安全。

导入 Path

from pathlib import Path

创建路径

from pathlib import Path

# 当前目录
current = Path(".")
print(current)  # 输出: .

# 用户主目录
home = Path.home()
print(home)  # 输出: /Users/username 或 C:\Users\username

# 拼接路径（用 / 运算符）
data_dir = Path("data")
file_path = data_dir / "test.json"
print(file_path)  # 输出: data/test.json

# 绝对路径
abs_path = Path("/Users/username/project/data.txt")
print(abs_path)

常用属性

from pathlib import Path

p = Path("data/test.json")

print(p.name)       # 输出: test.json（文件名）
print(p.stem)       # 输出: test（不带扩展名的文件名）
print(p.suffix)     # 输出: .json（扩展名）
print(p.parent)     # 输出: data（父目录）
print(p.exists())   # 输出: True 或 False（是否存在）
print(p.is_file())  # 输出: True 或 False（是否是文件）
print(p.is_dir())   # 输出: True 或 False（是否是目录）

常用操作

from pathlib import Path

# 创建目录（包括父目录）
Path("output/reports").mkdir(parents=True, exist_ok=True)

# 列出目录下的文件
for file in Path(".").iterdir():
    print(file.name)

# 列出所有 .py 文件
for file in Path(".").glob("*.py"):
    print(file.name)

# 递归列出所有 .py 文件
for file in Path(".").rglob("*.py"):
    print(file)

文本文件

读写文本文件是最基本的文件操作。

写入文本文件

from pathlib import Path

# 写入文本
content = """第一行：Hello
第二行：World
第三行：Python"""

Path("output.txt").write_text(content, encoding="utf-8")
print("文件已写入")

读取文本文件

from pathlib import Path

# 读取整个文件
content = Path("output.txt").read_text(encoding="utf-8")
print(content)

# 逐行读取
lines = Path("output.txt").read_text(encoding="utf-8").splitlines()
for i, line in enumerate(lines):
    print(f"第 {i+1} 行: {line}")

使用 with 语句（推荐）

# 写入文件
with open("output.txt", "w", encoding="utf-8") as f:
    f.write("第一行\n")
    f.write("第二行\n")
    f.write("第三行\n")

# 读取文件
with open("output.txt", "r", encoding="utf-8") as f:
    content = f.read()
    print(content)

# 逐行读取
with open("output.txt", "r", encoding="utf-8") as f:
    for line in f:
        print(line.strip())  # strip() 去掉换行符

追加内容

# 追加内容（不会覆盖原有内容）
with open("log.txt", "a", encoding="utf-8") as f:
    f.write("新的日志内容\n")

JSON 文件

JSON 是 LLM 应用中最常用的数据格式，Python 可以轻松读写 JSON 文件。

写入 JSON 文件

import json
from pathlib import Path

# 准备数据
data = {
    "project": "LLM 应用",
    "version": "1.0",
    "features": ["RAG", "Agent", "微调"],
    "team": [
        {"name": "张三", "role": "开发者"},
        {"name": "李四", "role": "测试"}
    ]
}

# 写入 JSON 文件
Path("config.json").write_text(
    json.dumps(data, ensure_ascii=False, indent=2),
    encoding="utf-8"
)
print("JSON 文件已写入")

读取 JSON 文件

import json
from pathlib import Path

# 读取 JSON 文件
content = Path("config.json").read_text(encoding="utf-8")
data = json.loads(content)

print(data["project"])    # 输出: LLM 应用
print(data["features"])   # 输出: ['RAG', 'Agent', '微调']

# 遍历团队成员
for member in data["team"]:
    print(f"{member['name']} - {member['role']}")

JSON Lines 格式

JSONL 是一种每行一个 JSON 对象的格式，常用于存储大量数据：

import json
from pathlib import Path

# 写入 JSONL
samples = [
    {"instruction": "什么是 RAG？", "output": "RAG 是检索增强生成"},
    {"instruction": "什么是 LLM？", "output": "LLM 是大语言模型"},
]

with open("samples.jsonl", "w", encoding="utf-8") as f:
    for sample in samples:
        f.write(json.dumps(sample, ensure_ascii=False) + "\n")

# 读取 JSONL
with open("samples.jsonl", "r", encoding="utf-8") as f:
    for line in f:
        data = json.loads(line.strip())
        print(data["instruction"], "->", data["output"])
与 LLM 应用的关联：JSONL 格式常用于存储微调数据和评估数据集。

CSV 文件

CSV（逗号分隔值）是一种简单的表格数据格式，常用于存储测试数据、评估结果等。

写入 CSV 文件

import csv
from pathlib import Path

# 准备数据
data = [
    ["case_id", "question", "expected_answer", "score"],
    ["TC001", "什么是 RAG？", "RAG 是检索增强生成", 0.95],
    ["TC002", "什么是 LLM？", "LLM 是大语言模型", 0.90],
    ["TC003", "什么是 Agent？", "Agent 是智能代理", 0.85],
]

# 写入 CSV
with open("test_results.csv", "w", encoding="utf-8", newline="") as f:
    writer = csv.writer(f)
    writer.writerows(data)

print("CSV 文件已写入")

读取 CSV 文件

import csv
from pathlib import Path

# 读取 CSV
with open("test_results.csv", "r", encoding="utf-8") as f:
    reader = csv.reader(f)
    for row in reader:
        print(row)

# 使用 DictReader（更方便）
with open("test_results.csv", "r", encoding="utf-8") as f:
    reader = csv.DictReader(f)
    for row in reader:
        print(f"案例 {row['case_id']}: {row['question']} -> 得分 {row['score']}")

使用 pandas（可选）

# 如果安装了 pandas，处理 CSV 更方便
import pandas as pd

# 读取 CSV
df = pd.read_csv("test_results.csv")
print(df)

# 筛选高分案例
high_scores = df[df["score"] > 0.9]
print(high_scores)

# 计算平均分
avg_score = df["score"].mean()
print(f"平均分: {avg_score}")
