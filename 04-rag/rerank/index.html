Rerank 重排序技术详解

Rerank是RAG系统中的关键技术，用于优化初始检索结果的排序，显著提升检索准确性和答案质量。

什么是Rerank？

Rerank（重排序）是信息检索中的第二阶段排序技术。在RAG系统中，它位于初始检索和最终生成之间，负责对检索结果进行精细化重新排序。

核心思想

粗筛 + 精排：先用快速方法召回候选文档，再用精确模型重新排序，平衡效率与准确性。

RAG流程中的位置

用户查询 → 初始检索 → [Rerank重排序] → 最终选择 → LLM生成
         ↓           ↓           ↓           ↓
    向量/关键词检索   Cross-Encoder   Top-K文档    答案生成
    (Top-50)        重新评分         (Top-5)

与初始检索的区别

特性 初始检索 Rerank
目的 快速召回候选文档 精细化重新排序
模型类型 Bi-Encoder（双塔模型） Cross-Encoder（交叉编码器）
速度 快（毫秒级） 慢（秒级）
精度 中等 高
计算复杂度 O(n) O(n²)

为什么需要Rerank？

初始检索的局限性

向量检索的问题

  • 语义相似但不精确匹配
  • 可能漏掉关键词完全匹配的文档
  • 对专业术语理解有限

关键词检索的问题

  • 无法理解同义词和近义词
  • 对语序和上下文不敏感
  • 可能返回不相关的精确匹配

Rerank带来的提升

实验数据（基于BGE-Reranker）

  • 准确率提升：10-30%（取决于数据集）
  • 召回率提升：15-25%
  • 答案质量：显著提高，减少幻觉
  • 用户满意度：提升20-40%

工作原理

1. Bi-Encoder vs Cross-Encoder

Bi-Encoder（双塔模型）

# 查询和文档分别编码
query_embedding = model.encode(query)      # [1, 768]
doc_embedding = model.encode(document)     # [1, 768]

# 计算相似度
score = cosine_similarity(query_embedding, doc_embedding)

# 特点：
# - 查询和文档独立编码
# - 可预计算文档向量
# - 速度快，适合大规模检索

Cross-Encoder（交叉编码器）

# 查询和文档拼接后一起编码
input_text = f"[CLS] {query} [SEP] {document} [SEP]"
score = model.predict(input_text)  # 直接输出相关性分数

# 特点：
# - 查询和文档交互编码
# - 无法预计算
# - 精度高，适合重排序

2. Cross-Encoder详细工作流程

from transformers import AutoTokenizer, AutoModelForSequenceClassification
import torch

class CrossEncoderReranker:
    def __init__(self, model_name="BAAI/bge-reranker-base"):
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModelForSequenceClassification.from_pretrained(model_name)
        self.model.eval()
    
    def rerank(self, query: str, documents: list[str], top_k: int = 5):
        """
        对文档进行重排序
        
        Args:
            query: 用户查询
            documents: 候选文档列表
            top_k: 返回前K个文档
        
        Returns:
            排序后的文档列表
        """
        scores = []
        
        for doc in documents:
            # 1. 拼接查询和文档
            inputs = self.tokenizer(
                query, 
                doc, 
                padding=True, 
                truncation=True, 
                max_length=512,
                return_tensors="pt"
            )
            
            # 2. 模型推理
            with torch.no_grad():
                outputs = self.model(**inputs)
                score = outputs.logits.item()
            
            scores.append(score)
        
        # 3. 按分数排序
        scored_docs = list(zip(documents, scores))
        scored_docs.sort(key=lambda x: x[1], reverse=True)
        
        return scored_docs[:top_k]

# 使用示例
reranker = CrossEncoderReranker()
query = "如何配置Embedding模型"
documents = [
    "Embedding模型配置需要设置维度、模型路径等参数。",
    "向量数据库用于存储和检索向量。",
    "配置文件位于config/settings.yaml。",
    "选择合适的Embedding模型对检索效果至关重要。"
]

results = reranker.rerank(query, documents, top_k=2)
for doc, score in results:
    print(f"Score: {score:.4f} | {doc[:50]}...")

3. 注意力机制的作用

Cross-Encoder通过交叉注意力机制，让查询和文档中的每个token都能相互关注，从而：

  • 捕捉查询词在文档中的精确位置
  • 理解词语之间的语义关系
  • 识别同义词和上下文关联

from sentence_transformers import CrossEncoder
import numpy as np

class SimpleReranker:
    def __init__(self, model_name="BAAI/bge-reranker-base"):
        self.model = CrossEncoder(model_name)
    
    def rerank(self, query: str, documents: list[str], top_k: int = 5):
        """简单重排序实现"""
        # 构建查询-文档对
        pairs = [[query, doc] for doc in documents]
        
        # 计算相关性分数
        scores = self.model.predict(pairs)
        
        # 排序并返回top_k
        scored_docs = list(zip(documents, scores))
        scored_docs.sort(key=lambda x: x[1], reverse=True)
        
        return scored_docs[:top_k]

2. 集成到LangChain

from langchain.retrievers import ContextualCompressionRetriever
from langchain.retrievers.document_compressors import CrossEncoderReranker
from langchain_community.cross_encoders import HuggingFaceCrossEncoder

class RAGWithRerank:
    def __init__(self, vector_store, reranker_model="BAAI/bge-reranker-base"):
        # 初始化向量检索器
        self.retriever = vector_store.as_retriever(
            search_type="similarity",
            search_kwargs={"k": 20}  # 初始检索20个文档
        )
        
        # 初始化Reranker
        reranker = HuggingFaceCrossEncoder(model_name=reranker_model)
        compressor = CrossEncoderReranker(model=reranker)
        
        # 创建带重排序的检索器
        self.compression_retriever = ContextualCompressionRetriever(
            base_compressor=compressor,
            base_retriever=self.retriever
        )
    
    def retrieve(self, query: str, top_k: int = 5):
        """检索并重排序"""
        compressed_docs = self.compression_retriever.invoke(query)
        return compressed_docs[:top_k]

# 使用示例
rag = RAGWithRerank(vector_store)

3. 完整的RAG Pipeline

from typing import List, Tuple
from dataclasses import dataclass
import chromadb
from sentence_transformers import SentenceTransformer, CrossEncoder

@dataclass
class Document:
    text: str
    metadata: dict
    score: float = 0.0

class RAGPipeline:
    def __init__(
        self,
        embedding_model: str = "BAAI/bge-small-zh-v1.5",
        reranker_model: str = "BAAI/bge-reranker-base",
        collection_name: str = "documents"
    ):
        # Embedding模型
        self.embedding_model = SentenceTransformer(embedding_model)
        
        # Reranker模型
        self.reranker = CrossEncoder(reranker_model)
        
        # 向量数据库
        self.chroma_client = chromadb.Client()
        self.collection = self.chroma_client.create_collection(
            name=collection_name,
            metadata={"hnsw:space": "cosine"}
        )
    
    def add_documents(self, documents: List[Document]):
        """添加文档到向量库"""
        texts = [doc.text for doc in documents]
        embeddings = self.embedding_model.encode(texts).tolist()
        
        self.collection.add(
            embeddings=embeddings,
            documents=texts,
            ids=[f"doc_{i}" for i in range(len(texts))],
            metadatas=[doc.metadata for doc in documents]
        )
    
    def initial_retrieve(self, query: str, top_k: int = 20) -> List[Document]:
        """初始检索"""
        query_embedding = self.embedding_model.encode(query).tolist()
        
        results = self.collection.query(
            query_embeddings=[query_embedding],
            n_results=top_k
        )
        
        documents = []
        for doc, metadata, distance in zip(
            results['documents'][0],
            results['metadatas'][0],
            results['distances'][0]
        ):
            documents.append(Document(
                text=doc,
                metadata=metadata,
                score=1 - distance  # 转换为相似度分数
            ))
        
        return documents
    
    def rerank_documents(
        self, 
        query: str, 
        documents: List[Document], 
        top_k: int = 5
    ) -> List[Document]:
        """重排序文档"""
        pairs = [[query, doc.text] for doc in documents]
        scores = self.reranker.predict(pairs)
        
        for doc, score in zip(documents, scores):
            doc.score = score
        
        # 按rerank分数排序
        documents.sort(key=lambda x: x.score, reverse=True)
        return documents[:top_k]
    
    def retrieve(
        self, 
        query: str, 
        initial_top_k: int = 20, 
        final_top_k: int = 5
    ) -> List[Document]:
        """完整的检索流程"""
        # 1. 初始检索
        initial_docs = self.initial_retrieve(query, initial_top_k)
        
        # 2. 重排序
        reranked_docs = self.rerank_documents(query, initial_docs, final_top_k)
        
        return reranked_docs

# 使用示例
pipeline = RAGPipeline()

# 添加文档
documents = [
    Document(text="RAG系统优化指南", metadata={"source": "guide.md"}),
    Document(text="Embedding模型选择", metadata={"source": "model.md"}),
    Document(text="向量数据库配置", metadata={"source": "config.md"}),
]
pipeline.add_documents(documents)

# 检索
results = pipeline.retrieve("如何优化RAG性能")
for doc in results:
    print(f"Score: {doc.score:.4f} | {doc.text}")

class HybridRAGWithRerank:
    """混合检索 + Rerank的完整方案"""
    
    def __init__(self):
        self.vector_retriever = VectorRetriever()
        self.bm25_retriever = BM25Retriever()
        self.reranker = CrossEncoderReranker()
    
    def retrieve(self, query: str, top_k: int = 5):
        # 1. 向量检索（语义相似）
        vector_results = self.vector_retriever.search(query, top_k=15)
        
        # 2. BM25检索（关键词匹配）
        bm25_results = self.bm25_retriever.search(query, top_k=15)
        
        # 3. 合并结果（去重）
        combined = self.merge_results(vector_results, bm25_results)
        
        # 4. Rerank精排
        reranked = self.reranker.rerank(query, combined, top_k=top_k)
        
        return reranked
    
    def merge_results(self, vector_results, bm25_results):
        """合并两种检索结果"""
        seen = set()
        combined = []
        
        for doc in vector_results + bm25_results:
            if doc.text not in seen:
                seen.add(doc.text)
                combined.append(doc)
        
        return combined

3. 缓存优化

from functools import lru_cache
import hashlib

class CachedReranker:
    """带缓存的Reranker"""
    
    def __init__(self, reranker):
        self.reranker = reranker
        self.cache = {}
    
    def _get_cache_key(self, query: str, documents: list[str]) -> str:
        """生成缓存键"""
        content = f"{query}|{'|'.join(documents)}"
        return hashlib.md5(content.encode()).hexdigest()
    
    def rerank(self, query: str, documents: list[str], top_k: int = 5):
        """带缓存的重排序"""
        cache_key = self._get_cache_key(query, documents)
        
        if cache_key in self.cache:
            return self.cache[cache_key]
        
        # 执行重排序
        results = self.reranker.rerank(query, documents, top_k)
        
        # 缓存结果
        self.cache[cache_key] = results
        
        return results

2. 批处理优化

class BatchReranker:
    """批处理Reranker"""
    
    def __init__(self, model, batch_size=32):
        self.model = model
        self.batch_size = batch_size
    
    def rerank_batch(self, query: str, documents: list[str], top_k: int = 5):
        """批量处理文档"""
        all_scores = []
        
        # 分批处理
        for i in range(0, len(documents), self.batch_size):
            batch = documents[i:i + self.batch_size]
            pairs = [[query, doc] for doc in batch]
            scores = self.model.predict(pairs)
            all_scores.extend(scores)
        
        # 排序
        scored_docs = list(zip(documents, all_scores))
        scored_docs.sort(key=lambda x: x[1], reverse=True)
        
        return scored_docs[:top_k]
